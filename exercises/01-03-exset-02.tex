\exercisesetinstructions[Use the following limits to evaluate the limits given in Exercises]{, where possible. If it is not possible, state so.
\begin{align*}
\lim_{x\to9}f(x) &= f(9) = 6 & \lim_{x\to6} f(x) &= f(6) = 9\\
\lim_{x\to9}g(x) &= g(9) = 3 & \lim_{x\to6} g(x) &= g(6) = 3
\end{align*}}

\exercise{$\ds \lim_{x\to9}(f(x)+g(x))$}{9}

\exercise{$\ds \lim_{x\to9}(3f(x)/g(x))$}{6}

\exercise{$\ds \lim_{x\to9}\left(\frac{f(x)-2g(x)}{g(x)}\right)$}{0}

\exercise{$\ds \lim_{x\to6}\left(\frac{f(x)}{3-g(x)}\right)$}{Limit does not exist.}

\exercise{$\ds \lim_{x\to9}g\big(f(x)\big)$}{3}

\exercise{$\ds \lim_{x\to6}f\big(g(x)\big)$}{Not possible to know.}

\exercise{$\ds \lim_{x\to6}g\big(f(f(x))\big)$}{3}

\exercise{$\ds \lim_{x\to6}\left(f(x)g(x)-f\,^2(x)+g^2(x)\right)$}{$-45$}

\exercisesetend
